cmake_minimum_required(VERSION 3.20)
project(PoissonJacobi VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(POISSON_ENABLE_MPI "Build poisson_mpi" ON)
option(POISSON_ENABLE_CUDA "Use the CUDA backend in poisson_mpi" ON)
include(CTest)
if(POISSON_ENABLE_CUDA AND NOT POISSON_ENABLE_MPI)
  message(FATAL_ERROR "The CUDA solver uses MPI; enable POISSON_ENABLE_MPI")
endif()
if(POISSON_ENABLE_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
endif()
if(POISSON_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
endif()
add_subdirectory(src)
add_subdirectory(executables)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
