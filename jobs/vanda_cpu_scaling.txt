#!/bin/bash
#PBS -N poisson_cpu_scale
#PBS -l select=4:ncpus=64:mpiprocs=64:ompthreads=1:mem=64gb
#PBS -l place=scatter
#PBS -l walltime=04:00:00
#PBS -j oe

set -eo pipefail
cd "${PBS_O_WORKDIR:?Submit from the project root}"
export POISSON_MODULE="${POISSON_MODULE:-nvhpc-hpcx-cuda12/26.1}"
export POISSON_BUILD_DIR="${POISSON_BUILD_DIR:-build-cpu}"
export POISSON_RESULTS_ROOT="${POISSON_RESULTS_ROOT:-results/vanda-cpu}"
source jobs/vanda_environment.sh
set -u

ranks="${POISSON_CPU_RANKS:-256}"
case "$ranks" in
    1) grid2d=1x1; grid3d=1x1x1 ;;
    4) grid2d=2x2; grid3d=2x2x1 ;;
    8) grid2d=2x4; grid3d=2x2x2 ;;
    16) grid2d=4x4; grid3d=4x2x2 ;;
    32) grid2d=4x8; grid3d=4x4x2 ;;
    64) grid2d=8x8; grid3d=4x4x4 ;;
    128) grid2d=8x16; grid3d=8x4x4 ;;
    256) grid2d=16x16; grid3d=8x8x4 ;;
    *) echo 'POISSON_CPU_RANKS must be 1, 4, 8, 16, 32, 64, 128, or 256.' >&2; exit 1 ;;
esac
ranks_per_node="${POISSON_CPU_RANKS_PER_NODE:-64}"
if [[ ! "$ranks_per_node" =~ ^[1-9][0-9]*$ ]]; then
    echo 'POISSON_CPU_RANKS_PER_NODE must be a positive integer.' >&2
    exit 1
fi
if (( ranks_per_node > ranks )); then ranks_per_node="$ranks"; fi
if (( ranks % ranks_per_node != 0 )); then
    echo 'Use the same number of ranks on each allocated CPU node.' >&2
    exit 1
fi
study="${POISSON_STUDY:-all}"
case "$study" in
    all|strong|weak) ;;
    *) echo 'POISSON_STUDY must be all, strong, or weak.' >&2; exit 1 ;;
esac

run_dir="$poisson_results/cpu-scale-${PBS_JOBID:?}"
mkdir -p "$run_dir"
exec > "$run_dir/job.log" 2>&1
module list
mpiexec --version
source jobs/vanda_cpu_launcher.sh
if command -v lscpu >/dev/null 2>&1; then lscpu; fi
if [[ -n "${PBS_NODEFILE:-}" ]]; then sort "$PBS_NODEFILE" | uniq -c; fi

# Reject accidental use of a CUDA build and verify the intended rank count.
IFS=x read -r px py pz <<< "$grid3d"
mpiexec -n "$ranks" "${poisson_mpi_args[@]}" \
    "$poisson_build/executables/poisson_mpi" \
    --dim 3 --nx 33 --ny 35 --nz 17 --px "$px" --py "$py" --pz "$pz" \
    --iterations 5 --check-every 0 --warmup 0 --overlap 1 \
    > "$run_dir/mapping.csv"
"$poisson_python" - "$run_dir/mapping.csv" "$ranks" <<'PY'
import sys
from pathlib import Path
from scripts.solver_output import parse_summary
row = parse_summary(Path(sys.argv[1]).read_text())
if (row['mode'] != 'mpi-cpu'
        or int(row['mpi_ranks']) != int(sys.argv[2]) or int(row['gpus_used']) != 0):
    raise SystemExit('Expected the CPU MPI build with the requested number of ranks.')
PY

# Larger reference grids support the 16x16 partition at 256 ranks.
"$poisson_python" tests/validate_mpi.py \
    --serial "$poisson_build/executables/poisson_serial" \
    --parallel "$poisson_build/executables/poisson_mpi" --backend cpu \
    --points 33 35 17 --grid2d "$grid2d" --grid3d "$grid3d" \
    "${poisson_launcher_args[@]}"

benchmark=("$poisson_python" scripts/benchmark.py \
    --exe "$poisson_build/executables/poisson_mpi" --mode mpi-cpu \
    --iterations "${POISSON_ITERATIONS:-1000}" --repeats "${POISSON_REPEATS:-3}" \
    --overlap 0 1 --communication host \
    "${poisson_launcher_args[@]}")

if [[ "$study" == all || "$study" == strong ]]; then
    "${benchmark[@]}" --kind strong --dim 2 --process-grids "$grid2d" \
        --nx "${POISSON_CPU_STRONG_2D:-4096}" --ny "${POISSON_CPU_STRONG_2D:-4096}" \
        --output "$run_dir/strong-2d.csv"
    "${benchmark[@]}" --kind strong --dim 3 --process-grids "$grid3d" \
        --nx "${POISSON_CPU_STRONG_3D:-256}" --ny "${POISSON_CPU_STRONG_3D:-256}" \
        --nz "${POISSON_CPU_STRONG_3D:-256}" --output "$run_dir/strong-3d.csv"
fi
if [[ "$study" == all || "$study" == weak ]]; then
    "${benchmark[@]}" --kind weak --dim 2 --process-grids "$grid2d" \
        --nx "${POISSON_CPU_WEAK_2D:-256}" --ny "${POISSON_CPU_WEAK_2D:-256}" \
        --output "$run_dir/weak-2d.csv"
    "${benchmark[@]}" --kind weak --dim 3 --process-grids "$grid3d" \
        --nx "${POISSON_CPU_WEAK_3D:-32}" --ny "${POISSON_CPU_WEAK_3D:-32}" \
        --nz "${POISSON_CPU_WEAK_3D:-32}" --output "$run_dir/weak-3d.csv"
fi
echo "Completed $ranks-core CPU MPI measurements in $run_dir"
