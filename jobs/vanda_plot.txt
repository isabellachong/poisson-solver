#!/bin/bash
#PBS -N poisson_figures
#PBS -l select=1:ncpus=2:mpiprocs=1:ompthreads=1:mem=8gb
#PBS -l walltime=00:15:00
#PBS -j oe

set -eo pipefail
cd "${PBS_O_WORKDIR:?Submit from the project root}"
source jobs/vanda_environment.sh
set -u

# Use a Python environment with requirements-analysis.txt installed.
shopt -s nullglob
inputs=("$poisson_results"/scale-*/strong-*.csv \
        "$poisson_results"/scale-*/weak-*.csv \
        "$poisson_results"/scale-*/sharing-*.csv \
        "$poisson_results"/cpu-scale-*/strong-*.csv \
        "$poisson_results"/cpu-scale-*/weak-*.csv \
        "$poisson_results"/serial-*/serial-*.csv)
if (( ${#inputs[@]} == 0 )); then
    echo "No completed benchmark CSVs in $poisson_results" >&2
    exit 1
fi
"$poisson_python" scripts/plot_scaling.py --input "${inputs[@]}" \
    --output-dir "$poisson_results/figures"
