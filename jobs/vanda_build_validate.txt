#!/bin/bash
#PBS -N poisson_check
#PBS -l select=1:ncpus=36:mpiprocs=8:ompthreads=1:ngpus=1
#PBS -l walltime=01:00:00
#PBS -j oe

set -eo pipefail
cd "${PBS_O_WORKDIR:?Submit from the project root}"
source jobs/vanda_environment.sh
set -u

log_dir="$poisson_results/check-${PBS_JOBID:?}"
mkdir -p "$log_dir"
exec > "$log_dir/job.log" 2>&1
module list
source jobs/vanda_build_environment.sh
nvidia-smi
nvcc --version
mpiexec --version

# native requires CMake >=3.24 and detects the GPU in this allocation.
cmake -S . -B "$poisson_build" -DCMAKE_BUILD_TYPE=Release \
    -DPOISSON_ENABLE_MPI=ON -DPOISSON_ENABLE_CUDA=ON \
    "-DMPI_CXX_COMPILER=$(command -v mpicxx)" \
    "-DCMAKE_CUDA_ARCHITECTURES=${POISSON_CUDA_ARCHITECTURES:-native}"
cmake --build "$poisson_build" --parallel 4
ctest --test-dir "$poisson_build" --output-on-failure

# One allocated GPU must permit eight MPI processes to share it.
"$poisson_python" tests/validate_mpi.py \
    --serial "$poisson_build/executables/poisson_serial" \
    --parallel "$poisson_build/executables/poisson_mpi" --backend cuda \
    --communication "${POISSON_COMMUNICATION:-both}" \
    --launcher-arg=--map-by --launcher-arg=ppr:8:node \
    --launcher-arg=--bind-to "--launcher-arg=$poisson_bind_to"
