#!/bin/bash
#PBS -N serial_figures
#PBS -l select=1:ncpus=1:mpiprocs=1:ompthreads=1:mem=2gb
#PBS -l walltime=00:10:00
#PBS -j oe

set -eo pipefail
cd "${PBS_O_WORKDIR:?Submit from the project root}"
export POISSON_PYTHON="${POISSON_PYTHON:-$PWD/.venv-analysis/bin/python}"
source jobs/vanda_environment.sh
set -u

serial_dir="${POISSON_SERIAL_RUN:?Set POISSON_SERIAL_RUN to a completed serial results directory}"
for input in "$serial_dir/serial-2d.csv" "$serial_dir/serial-3d.csv"; do
    if [[ ! -s "$input" ]]; then
        echo "Missing or empty serial benchmark CSV: $input" >&2
        exit 1
    fi
done

# Read only this completed serial run; no parallel-job dependencies are needed.
"$poisson_python" scripts/plot_serial.py \
    --input "$serial_dir/serial-2d.csv" "$serial_dir/serial-3d.csv" \
    --output-dir "$serial_dir/figures"
echo "Completed serial figures in $serial_dir/figures"
