#!/bin/bash
#PBS -N poisson_serial
#PBS -l select=1:ncpus=1:mpiprocs=1:ompthreads=1:mem=4gb
#PBS -l walltime=04:00:00
#PBS -j oe

set -eo pipefail
cd "${PBS_O_WORKDIR:?Submit from the project root}"
export POISSON_MODULE="${POISSON_MODULE:-nvhpc-hpcx-cuda12/26.1}"
export POISSON_BUILD_DIR="${POISSON_BUILD_DIR:-build-cpu}"
export POISSON_RESULTS_ROOT="${POISSON_RESULTS_ROOT:-results/vanda-cpu}"
source jobs/vanda_environment.sh
set -u

run_dir="$poisson_results/serial-${PBS_JOBID:?}"
mkdir -p "$run_dir"
exec > "$run_dir/job.log" 2>&1
module list
hostname
if command -v lscpu >/dev/null 2>&1; then lscpu; fi
"$poisson_python" tests/validate_mpi.py --serial-only \
    --serial "$poisson_build/executables/poisson_serial"

benchmark=("$poisson_python" scripts/benchmark.py \
    --exe "$poisson_build/executables/poisson_serial" --mode serial --kind serial \
    --iterations "${POISSON_ITERATIONS:-1000}" --repeats "${POISSON_REPEATS:-3}")

# Equal-unknown pairs include 64^2=16^3, 512^2=64^3, and 4096^2=256^3.
"${benchmark[@]}" --dim 2 --sizes 64 128 256 512 1024 2048 4096 \
    --output "$run_dir/serial-2d.csv"
"${benchmark[@]}" --dim 3 --sizes 16 32 64 128 256 \
    --output "$run_dir/serial-3d.csv"
echo "Completed serial size sweeps in $run_dir"
